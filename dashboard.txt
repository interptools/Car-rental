function parseRecordTimestamp(value) {
  if (value === undefined || value === null || value === "") return null;
  if (typeof value === "string" && /^\d+$/.test(value.trim())) value = Number(value.trim());
  if (typeof value === "number") { const ms = value > 1e12 ? value : value * 1000; const d = new Date(ms); return isNaN(d.getTime()) ? null : d; }
  if (typeof value === "string") { const d = new Date(value); return isNaN(d.getTime()) ? null : d; }
  return null;
}
function escapeHtml(v) { return String(v ?? "").replace(/[&<>"']/g, ch => ({ "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;" }[ch])); }
function fmt(v) { const d = parseRecordTimestamp(v); return d ? d.toLocaleString("en-PH", { timeZone: PH_TIMEZONE }) : "—"; }
function formatDate(v) { const d = parseRecordTimestamp(v); return d ? d.toLocaleDateString("en-PH", { timeZone: PH_TIMEZONE, year: "numeric", month: "long", day: "numeric" }) : "—"; }
function formatDateTime(v) {
  const d = parseRecordTimestamp(v); if (!d) return "—";
  const dp = d.toLocaleDateString("en-PH", { timeZone: PH_TIMEZONE, year: "numeric", month: "long", day: "numeric" });
  const tp = d.toLocaleTimeString("en-PH", { timeZone: PH_TIMEZONE, hour: "numeric", minute: "2-digit", hour12: true });
  return `${dp} at ${tp}`;
}
function period(r) { return `${formatDateTime(r.pickupDate) || "—"} → ${formatDateTime(r.returnDate) || "—"}`; }
function statusOf(r) { const s = String(r.status || "").trim(); return s ? s : "Pending"; }
function parseTermsToArray(t) {
  if (!t) return TERMS;
  if (Array.isArray(t)) return t;
  if (typeof t === "string") { try { const p = JSON.parse(t); if (Array.isArray(p)) return p; } catch(e){} return [["Terms and Conditions", t]]; }
  return TERMS;
}

/* Load */
async function loadRecords() {
  const status = $("loadStatus");
  status.textContent = "Loading records…";
  if (APPS_SCRIPT_URL.includes("PASTE_YOUR")) { status.textContent = "Setup required."; return; }
  try {
    const result = await requestRecordsJsonp();
    if (!result.ok) throw new Error(result.error || "The server rejected the request.");
    records = Array.isArray(result.records) ? result.records : [];
    renderRecords();
    status.textContent = `Loaded ${records.length} record(s) at ${new Date().toLocaleString("en-PH", { timeZone: PH_TIMEZONE })}.`;
  } catch (error) {
    console.error(error);
    status.textContent = "Could not load records. Verify URL, deployment, key, and browser console.";
  }
}
function requestRecordsJsonp() {
  return new Promise((resolve, reject) => {
    const cb = "__rentalRecordsCallback";
    const prev = window[cb];
    const script = document.createElement("script");
    const url = new URL(APPS_SCRIPT_URL);
    let done = false;
    const cleanup = () => { if (script.parentNode) script.parentNode.removeChild(script); if (prev === undefined) delete window[cb]; else window[cb] = prev; };
    const t = setTimeout(() => { if (done) return; done = true; cleanup(); reject(new Error("timeout")); }, 20000);
    window[cb] = r => { if (done) return; done = true; clearTimeout(t); cleanup(); resolve(r); };
    script.onerror = () => { if (done) return; done = true; clearTimeout(t); cleanup(); reject(new Error("network")); };
    url.searchParams.set("action", "list");
    url.searchParams.set("key", adminKey);
    url.searchParams.set("callback", cb);
    url.searchParams.set("_", String(Date.now()));
    script.src = url.toString();
    document.head.appendChild(script);
  });
}

    const url = new URL(APPS_SCRIPT_URL);
    let done = false;

    url.searchParams.set("action", action);

function renderRecords() {
  const q = $("searchInput").value.trim().toLowerCase();
  const filtered = records.filter(r => [r.fullName, r.contact, r.email, r.vehiclePlate, r.licenseNo].some(v => String(v || "").toLowerCase().includes(q)));
  $("totalRecords").textContent = records.length;
  $("acceptedCount").textContent = records.filter(r => String(r.agreementAccepted).toLowerCase() === "yes").length;
  const latest = records[records.length - 1];
  $("latestRecord").textContent = latest ? fmt(latest.timestamp || latest.submittedAt) : "—";
  const body = $("recordsBody");
  if (!filtered.length) {
    body.innerHTML = `<tr><td colspan="9" class="text-slate-500">${records.length ? "No records match your search." : "No records found."}</td></tr>`;
    return;
  }
  body.innerHTML = filtered.map(r => {
    const idx = records.indexOf(r);
    const ts = r.timestamp || r.submittedAt;
    const status = statusOf(r);
    const statusBadge = status === "Confirmed"
      ? `<span class="rounded-full px-2 py-1 text-xs bg-emerald-100 text-emerald-800">CONFIRMED</span>`
      : `<span class="rounded-full px-2 py-1 text-xs bg-amber-100 text-amber-800">PENDING</span>`;
    const accepted = String(r.agreementAccepted || "").toLowerCase() === "yes";
    const acceptedBadge = accepted
      ? `<span class="rounded-full px-2 py-1 text-xs bg-green-100 text-green-800">YES</span>`
      : `<span class="rounded-full px-2 py-1 text-xs bg-red-100 text-red-800">NO</span>`;
    const confirmLabel = status === "Confirmed" ? "Unconfirm" : "Confirm";
    const confirmClass = status === "Confirmed" ? "bg-amber-600 hover:bg-amber-700" : "bg-emerald-600 hover:bg-emerald-700";
    return `<tr>
      <td>${escapeHtml(fmt(ts))}</td>
      <td><strong>${escapeHtml(r.fullName || "—")}</strong><div class="text-xs text-slate-500">${escapeHtml(r.email || "")}</div></td>
      <td>${escapeHtml(r.contact || "—")}</td>
      <td>${escapeHtml(period(r))}</td>
      <td>${escapeHtml(r.totalDays || "—")}</td>
      <td>${escapeHtml(r.vehicleModel || VEHICLE.model)}<div class="text-xs text-slate-500">${escapeHtml(r.vehiclePlate || VEHICLE.plate)} · ${escapeHtml(r.vehicleColor || VEHICLE.color)}</div></td>
      <td>${acceptedBadge}</td>
      <td>${statusBadge}</td>
      <td>
        <div class="flex flex-col gap-1.5">
          <button class="pdfBtn rounded-lg bg-slate-900 hover:bg-slate-700 text-white px-3 py-1.5 text-xs whitespace-nowrap" data-index="${idx}">PDF</button>
          <button class="confirmBtn rounded-lg ${confirmClass} text-white px-3 py-1.5 text-xs whitespace-nowrap" data-row="${r._row}" data-confirmed="${status === "Confirmed"}">${confirmLabel}</button>
          <button class="emailBtn rounded-lg border border-slate-300 hover:bg-slate-50 px-3 py-1.5 text-xs whitespace-nowrap" data-index="${idx}">Email</button>
        </div>
      </td>
    </tr>`;
  }).join("");
  body.querySelectorAll(".pdfBtn").forEach(b => b.addEventListener("click", () => {
    const rec = records[Number(b.dataset.index)]; if (rec) downloadAgreementPdf(rec);
  }));
  body.querySelectorAll(".emailBtn").forEach(b => b.addEventListener("click", () => {
    const rec = records[Number(b.dataset.index)]; if (rec) openEmailModal(rec);
  }));
  body.querySelectorAll(".confirmBtn").forEach(b => b.addEventListener("click", () => {
    const row = Number(b.dataset.row);
    const rec = records.find(x => Number(x._row) === row);
    if (rec) toggleConfirm(rec, b.dataset.confirmed === "true");
  }));
}

async function toggleConfirm(record, isConfirmed) {
  const name = record.fullName || "this renter";
  const action = isConfirmed ? "cancel" : "confirm";
  const verb = isConfirmed ? "Unconfirm" : "Confirm";
  if (!confirm(`${verb} the reservation for ${name}?`)) return;
  if (!isConfirmed && String(record.agreementAccepted || "").toLowerCase() !== "yes") {
    if (!confirm(`This record did not record agreement acceptance. Continue anyway?`)) return;
  }
  try {
    const result = await requestAdminAction(action, record._row);
    if (!result.ok) { alert(`${verb} failed: ${result.error || "unknown error"}`); return; }
    record.status = action === "confirm" ? "Confirmed" : "Pending";
    renderRecords();
    if (action === "confirm") openEmailModal(record);
    loadRecords();
  } catch (err) { alert(`${verb} failed: ${err.message}`); }
}

/* PDF */
function downloadAgreementPdf(r) {
  if (!window.jspdf || !window.jspdf.jsPDF) { alert("PDF library did not load. Check your connection."); return; }
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF({ unit: "mm", format: "a4" });
  const pw = doc.internal.pageSize.getWidth(), ph = doc.internal.pageSize.getHeight();
  const margin = 14, maxWidth = pw - margin * 2, labelWidth = 56;
  let y = 18;
  function ensureSpace(req = 10) { if (y + req > ph - 14) { doc.addPage(); y = 18; } }
  function addField(label, value) {
    ensureSpace(8);
    doc.setFont("helvetica", "bold"); doc.setFontSize(8); doc.text(`${label}:`, margin, y);
    doc.setFont("helvetica", "normal"); doc.setFontSize(8);
    const lines = doc.splitTextToSize(String(value ?? "—"), maxWidth - labelWidth - 4);
    doc.text(lines, margin + labelWidth, y); y += Math.max(5, lines.length * 4);
  }
  function addParagraph(text) {
    const lines = doc.splitTextToSize(String(text ?? "—"), maxWidth);
    ensureSpace(lines.length * 4 + 4);
    doc.setFont("helvetica", "normal"); doc.setFontSize(8);
    doc.text(lines, margin, y); y += lines.length * 4 + 4;
  }
  function addSectionTitle(t) {
    ensureSpace(12);
    doc.setFont("helvetica", "bold"); doc.setFontSize(10); doc.text(t, margin, y); y += 6;
    doc.setDrawColor(203, 213, 225); doc.line(margin, y - 2, pw - margin, y - 2); y += 4;
  }
  function addSignatureBlock() {
    const sy = y + 6, lx = margin, rx = pw - margin - 70, len = 70;
    doc.setFont("helvetica", "normal"); doc.setFontSize(8);
    doc.line(lx, sy, lx + len, sy); doc.line(rx, sy, rx + len, sy);
    doc.text("Renter signature / date", lx, sy + 6);
    doc.text("Owner / representative signature / date", rx, sy + 6);
  }
  function getTs(rec) { return rec.timestamp || rec.submittedAt; }

  doc.setFillColor(15, 23, 42); doc.rect(0, 0, pw, 34, "F");
  doc.setTextColor(255, 255, 255); doc.setFont("helvetica", "bold"); doc.setFontSize(16);
  doc.text("VEHICLE RENTAL AGREEMENT", margin, 16);
  doc.setFont("helvetica", "normal"); doc.setFontSize(9);
  doc.text(`${r.vehicleModel || VEHICLE.model} | Plate ${r.vehiclePlate || VEHICLE.plate} | ${r.vehicleColor || VEHICLE.color}`, margin, 26);
  doc.setTextColor(0, 0, 0);
  y = 40;
  addSectionTitle("1. RENTER INFORMATION");
  addField("Full legal name", r.fullName); addField("Contact number", r.contact);
  addField("Address", r.address); addField("Email", r.email);
  addField("Date of birth", formatDate(r.dob)); addField("Driver's licence number", r.licenseNo);
  addField("Licence expiry", formatDate(r.licenseExpiry));
  addField("Government ID type", r.govIdType); addField("Government ID number", r.govIdNumber);
  addField("Emergency contact", `${r.emergencyName || "—"} (${r.emergencyPhone || "—"})`);
  y += 2; addSectionTitle("2. RENTAL AND VEHICLE CONDITION");
  addField("Vehicle", `${r.vehicleModel || VEHICLE.model}; ${r.vehicleColor || VEHICLE.color}; Plate ${r.vehiclePlate || VEHICLE.plate}`);
  addField("Pick-up date/time", formatDateTime(r.pickupDate));
  addField("Return date/time", formatDateTime(r.returnDate));
  addField("Total rental days", r.totalDays); addField("Odometer out (km)", r.odometerOut);
  addField("Fuel level out", r.fuelLevelOut); addField("Pre-existing damage / notes", r.notes);
  y += 2; addSectionTitle("3. TERMS AND CONDITIONS");
  parseTermsToArray(r.terms || null).forEach(([title, text]) => {
    doc.setFont("helvetica", "bold"); doc.setFontSize(9); doc.text(title, margin, y); y += 5; addParagraph(text);
  });
  y += 2; addSectionTitle("4. ACKNOWLEDGEMENT");
  addParagraph(`Agreement accepted: ${String(r.agreementAccepted || "No").toUpperCase()}`);
  addParagraph("The renter's electronic submission records the acknowledgement selected on the form. This generated copy is a record of the submitted data and is not a substitute for any physical copy or proof of vehicle handover.");
  addParagraph(`Record timestamp: ${formatDateTime(getTs(r))}  |  Terms version: ${String(r.termsVersion || "1.0")}  |  Status: ${statusOf(r)}`);
  addSignatureBlock();

  const safeName = String(r.fullName || "renter").replace(/[^a-z0-9_-]+/gi, "_").slice(0, 50);
  const ts = parseRecordTimestamp(getTs(r));
  const fileDate = ts ? ts.toLocaleDateString("en-CA", { timeZone: PH_TIMEZONE }) : "record";
  doc.save(`Rental_Agreement_${safeName}_${fileDate}.pdf`);
}
